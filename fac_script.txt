// ── Carga ──────────────────────────────────────────────
async function cargarFacturas() {
  try {
    const rows = await sg('facturas', 'order=fecha.desc&limit=500');
    G.facturas = rows;
    filtrarFacturas();
  } catch(err) {
    toast('Error cargando facturas: ' + err.message, 'err');
    G.facturas = [];
    filtrarFacturas();
  }
}

// ── Filtrado y render tabla ────────────────────────────
function filtrarFacturas() {
  if (!G.facturas) G.facturas = [];
  const q     = (document.getElementById('fac-busq')?.value || '').toLowerCase();
  const desde = document.getElementById('fac-desde')?.value || '';
  const hasta = document.getElementById('fac-hasta')?.value || '';
  const est   = document.getElementById('fac-estado')?.value || '';

  facFiltradas = G.facturas.filter(f => {
    const txt = ((f.numero_factura||'') + ' ' + nomPacFull(f.id_paciente) + ' ' + (f.receptor_nombre||'')).toLowerCase();
    if (q && !txt.includes(q)) return false;
    if (desde && f.fecha < desde) return false;
    if (hasta && f.fecha > hasta) return false;
    if (est && f.estado !== est) return false;
    return true;
  });

  const anio = new Date().getFullYear();
  const totalAnio = G.facturas
    .filter(f => f.fecha && f.fecha.startsWith(anio) && f.estado !== 'Anulada')
    .reduce((s, f) => s + (parseFloat(f.total) || 0), 0);
  const pendiente = G.facturas.filter(f => f.estado === 'Emitida').length;

  document.getElementById('fac-total-anio').textContent = fmtEur(totalAnio);
  document.getElementById('fac-pendiente').textContent  = pendiente + ' factura' + (pendiente !== 1 ? 's' : '');
  document.getElementById('fac-count').textContent      = G.facturas.filter(f => f.estado !== 'Anulada').length;

  facPag = 0;
  _renderFacturasTable();
}

function _renderFacturasTable() {
  const inicio = facPag * G.paginaSize;
  const pagina = facFiltradas.slice(inicio, inicio + G.paginaSize);
  const tbody  = document.getElementById('tbody-facturas');

  if (!pagina.length) {
    tbody.innerHTML = '<tr><td colspan="9" class="empty-state">Sin facturas</td></tr>';
  } else {
    tbody.innerHTML = pagina.map(f => `
      <tr onclick="editarFactura('${f.id_factura}')">
        <td class="mono" style="font-weight:600">${f.numero_factura || '—'}</td>
        <td>${fmtFecha(f.fecha)}</td>
        <td class="strong">${nomPacFull(f.id_paciente)}</td>
        <td class="muted">${f.receptor_nombre || '—'}</td>
        <td class="mono">${fmtEur(f.base_imponible)}</td>
        <td class="muted">${(parseFloat(f.iva_pct)||0) > 0 ? fmtEur(f.iva_importe) : '—'}</td>
        <td class="mono" style="font-weight:600;color:var(--blue)">${fmtEur(f.total)}</td>
        <td>${badgeEstado(f.estado === 'Pagada' ? 'Cobrado' : f.estado === 'Anulada' ? 'Cancelada' : 'Programada')}</td>
        <td style="display:flex;gap:6px">
          <button class="btn btn-secondary btn-sm" onclick="event.stopPropagation();editarFactura('${f.id_factura}')">✏️</button>
          <button class="btn btn-secondary btn-sm" onclick="event.stopPropagation();descargarPDFFactura('${f.id_factura}')" title="Descargar PDF">⬇ PDF</button>
        </td>
      </tr>`).join('');
  }
  document.getElementById('fac-footer').textContent = facFiltradas.length + ' facturas · Total: ' + fmtEur(facFiltradas.reduce((s,f)=>s+(parseFloat(f.total)||0),0));
  renderPaginacion('pag-facturas', facFiltradas.length, G.paginaSize, facPag, (p) => { facPag = p; _renderFacturasTable(); });
}

function recalcularTotales() {
  const base  = facLineas.reduce((s, l) => s + (parseFloat(l.importe) || 0), 0);
  const ivaPct  = parseFloat(document.getElementById('fac-iva-pct')?.value) || 0;
  const irpfPct = parseFloat(document.getElementById('fac-irpf-pct')?.value) || 0;
  const iva     = base * ivaPct / 100;
  const irpf    = base * irpfPct / 100;
  const total   = base + iva - irpf;
  const fmt     = v => v.toLocaleString('es-ES', {minimumFractionDigits:2, maximumFractionDigits:2}) + ' €';
  const el = id => document.getElementById(id);
  if (el('fac-calc-base'))  el('fac-calc-base').textContent  = fmt(base);
  if (el('fac-calc-iva'))   el('fac-calc-iva').textContent   = fmt(iva);
  if (el('fac-calc-irpf'))  el('fac-calc-irpf').textContent  = fmt(irpf);
  if (el('fac-calc-total')) el('fac-calc-total').textContent = fmt(total);
}

function escHtml(str) {
  return (str||'').replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
}

// ── Cargar líneas desde sesiones del paciente ──────────
async function cargarLineasDesdeSesiones() {
  const pacId = document.getElementById('fac-pac-id').value;
  const desde = document.getElementById('fac-ses-desde').value;
  const hasta = document.getElementById('fac-ses-hasta').value;
  if (!pacId) { toast('Selecciona primero un paciente', 'err'); return; }
  try {
    let params = `id_paciente=eq.${pacId}&estado=eq.Hecha`;
    if (desde) params += `&fecha=gte.${desde}`;
    if (hasta) params += `&fecha=lte.${hasta}`;
    const sesiones = await sg('sesiones', params + '&order=fecha.asc');
    if (!sesiones.length) { toast('No hay sesiones Hecha en ese período', 'err'); return; }

    // Agrupa por terapeuta+servicio para crear líneas limpias
    const grupos = {};
    sesiones.forEach(s => {
      const key = (s.id_terapeuta||'') + '|' + (s.id_servicio||'');
      if (!grupos[key]) grupos[key] = { sesiones: [], importe: 0, key };
      grupos[key].sesiones.push(s);
      grupos[key].importe += parseFloat(s.importe) || 0;
    });

    facLineas = Object.values(grupos).map(g => {
      const count = g.sesiones.length;
      const total = g.importe;
      const precio = count > 0 ? total / count : 45;
      const ter = nomTer(g.sesiones[0].id_terapeuta);
      const ses0 = g.sesiones[0];
      const sesN = g.sesiones[g.sesiones.length - 1];
      const rango = fmtFecha(ses0.fecha) + (count > 1 ? ' al ' + fmtFecha(sesN.fecha) : '');
      const concepto = `Asistencia psicológica · ${count} sesión${count>1?'es':''} de ${ter} (${rango})`;
      return { concepto, horas: count, precio_hora: precio, importe: total };
    });

    renderLineas();
    toast(sesiones.length + ' sesiones cargadas ✓', 'ok');
  } catch(err) { toast('Error: ' + err.message, 'err'); }
}

// ── Autocompletar datos receptor al seleccionar paciente ─
function rellenarDatosReceptor() {
  // Se llama también desde el onblur del pac-search via setTimeout
  setTimeout(() => {
    const id = document.getElementById('fac-pac-id').value;
    if (!id) return;
    const p = G.pacientes.find(x => x.id_paciente === id);
    if (!p) return;
    // Usa tutor1 si existe, si no el propio paciente
    const nombre = p.tutor1 || ((p.nombre||'') + ' ' + (p.apellidos||'')).trim();
    const dni    = p.dni_tutor1 || p.dni_paciente || '';
    const dir    = p.direccion || '';
    const cp     = p.codigo_postal || '';
    const mun    = p.municipio || '';
    const el = id2 => document.getElementById(id2);
    if (el('fac-rec-nombre') && !el('fac-rec-nombre').value) el('fac-rec-nombre').value = nombre;
    if (el('fac-rec-dni')    && !el('fac-rec-dni').value)    el('fac-rec-dni').value    = dni;
    if (el('fac-rec-dir')    && !el('fac-rec-dir').value)    el('fac-rec-dir').value    = dir;
    if (el('fac-rec-cp')     && !el('fac-rec-cp').value)     el('fac-rec-cp').value     = cp;
    if (el('fac-rec-mun')    && !el('fac-rec-mun').value)    el('fac-rec-mun').value    = mun;
  }, 300);
}

// ── Guardar ────────────────────────────────────────────
function _recogerDatosFactura() {
  const base    = facLineas.reduce((s, l) => s + (parseFloat(l.importe) || 0), 0);
  const ivaPct  = parseFloat(document.getElementById('fac-iva-pct').value) || 0;
  const irpfPct = parseFloat(document.getElementById('fac-irpf-pct').value) || 0;
  const iva     = base * ivaPct / 100;
  const irpf    = base * irpfPct / 100;
  return {
    numero_factura:    document.getElementById('fac-numero').value.trim(),
    fecha:             document.getElementById('fac-fecha').value,
    id_paciente:       document.getElementById('fac-pac-id').value || null,
    receptor_nombre:   document.getElementById('fac-rec-nombre').value.trim(),
    receptor_dni:      document.getElementById('fac-rec-dni').value.trim(),
    receptor_direccion:document.getElementById('fac-rec-dir').value.trim(),
    receptor_cp:       document.getElementById('fac-rec-cp').value.trim(),
    receptor_municipio:document.getElementById('fac-rec-mun').value.trim(),
    lineas:            facLineas,
    base_imponible:    parseFloat(base.toFixed(2)),
    iva_pct:           ivaPct,
    iva_importe:       parseFloat(iva.toFixed(2)),
    irpf_pct:          irpfPct,
    irpf_importe:      parseFloat(irpf.toFixed(2)),
    total:             parseFloat((base + iva - irpf).toFixed(2)),
    estado:            document.getElementById('fac-estado-modal').value,
    notas:             document.getElementById('fac-notas').value.trim(),
  };
}

async function guardarFactura() {
  const id    = document.getElementById('fac-id').value;
  const datos = _recogerDatosFactura();
  if (!datos.numero_factura || !datos.fecha) { toast('Número y fecha son obligatorios', 'err'); return; }
  if (!facLineas.length) { toast('Añade al menos una línea de concepto', 'err'); return; }
  try {
    if (id) {
      await sa('facturas', `id_factura=eq.${id}`, datos);
      const idx = G.facturas.findIndex(f => f.id_factura === id);
      if (idx >= 0) G.facturas[idx] = { ...G.facturas[idx], ...datos };
      toast('Factura actualizada ✓', 'ok');
    } else {
      const newId = await genId('facturas', 'id_factura', 'FAC');
      await sp('facturas', { ...datos, id_factura: newId });
      G.facturas.unshift({ ...datos, id_factura: newId });
      toast('Factura guardada ✓', 'ok');
    }
    cerrarModal('modal-factura');
    filtrarFacturas();
  } catch(err) { toast('Error: ' + err.message, 'err'); }
}

async function guardarYDescargarFactura() {
  const id    = document.getElementById('fac-id').value;
  const datos = _recogerDatosFactura();
  if (!datos.numero_factura || !datos.fecha) { toast('Número y fecha son obligatorios', 'err'); return; }
  if (!facLineas.length) { toast('Añade al menos una línea', 'err'); return; }
  try {
    let facId = id;
    if (id) {
      await sa('facturas', `id_factura=eq.${id}`, datos);
      const idx = G.facturas.findIndex(f => f.id_factura === id);
      if (idx >= 0) G.facturas[idx] = { ...G.facturas[idx], ...datos };
    } else {
      const newId = await genId('facturas', 'id_factura', 'FAC');
      facId = newId;
      await sp('facturas', { ...datos, id_factura: newId });
      G.facturas.unshift({ ...datos, id_factura: newId });
    }
    cerrarModal('modal-factura');
    filtrarFacturas();
    toast('Guardando PDF…');
    setTimeout(() => generarPDFFactura({ id_factura: facId, ...datos }, true), 300);
  } catch(err) { toast('Error: ' + err.message, 'err'); }
}

async function eliminarFactura() {
  const id = document.getElementById('fac-id').value; if (!id) return;
  if (!confirm('¿Eliminar factura ' + id + '? Esta acción no se puede deshacer.')) return;
  try {
    await sd('facturas', `id_factura=eq.${id}`);
    G.facturas = G.facturas.filter(f => f.id_factura !== id);
    toast('Factura eliminada ✓', 'ok');
    cerrarModal('modal-factura');
    filtrarFacturas();
  } catch(err) { toast('Error: ' + err.message, 'err'); }
}

function generarPDFFactura(f, descargar = true) {
  if (typeof jspdf === 'undefined' && typeof window.jspdf === 'undefined') {
    toast('jsPDF no cargado — revisa el CDN en <head>', 'err');
    return;
  }
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF({ orientation: 'portrait', unit: 'mm', format: 'a4' });

  const W = 210, MARGIN = 18;
  const BLUE = [16, 6, 159];
  const ORANGE = [241, 142, 28];
  const DARK = [26, 31, 53];
  const MUTED = [107, 116, 144];
  const BORDER = [228, 232, 240];
  const CREAM = [242, 244, 248];

  const fmt2 = v => (parseFloat(v)||0).toLocaleString('es-ES', {minimumFractionDigits:2, maximumFractionDigits:2}) + ' €';
  const fmt2plain = v => (parseFloat(v)||0).toLocaleString('es-ES', {minimumFractionDigits:2, maximumFractionDigits:2});

  // ── Cabecera azul top ──────────────────────────────
  doc.setFillColor(...BLUE);
  doc.rect(0, 0, W, 2, 'F');

  // ── Logo / nombre institución (izquierda) ──────────
  doc.setFont('helvetica', 'bold');
  doc.setFontSize(15);
  doc.setTextColor(...BLUE);
  doc.text('IPSE', MARGIN, 20);

  doc.setFont('helvetica', 'normal');
  doc.setFontSize(8);
  doc.setTextColor(...MUTED);
  doc.text('Instituto Psicoeducativo', MARGIN, 25);

  // ── Datos emisor (derecha) ─────────────────────────
  const exr = W - MARGIN;
  doc.setFont('helvetica', 'bold');
  doc.setFontSize(8);
  doc.setTextColor(...DARK);
  doc.text(EMISOR.nombre, exr, 10, { align: 'right' });
  doc.setFont('helvetica', 'normal');
  doc.setFontSize(7.5);
  doc.setTextColor(...MUTED);
  [
    EMISOR.nica,
    EMISOR.persona,
    EMISOR.master,
    EMISOR.ncol,
    EMISOR.nif,
    EMISOR.dir1,
    EMISOR.dir2 + ' · ' + EMISOR.dir3,
    EMISOR.tlf + ' · ' + EMISOR.email,
  ].forEach((line, i) => doc.text(line, exr, 15 + i * 4.2, { align: 'right' }));

  // ── Línea separadora ──────────────────────────────
  doc.setDrawColor(...BORDER);
  doc.setLineWidth(0.3);
  doc.line(MARGIN, 52, W - MARGIN, 52);

  // ── Fecha y número factura ─────────────────────────
  const fechaStr = (() => {
    if (!f.fecha) return '—';
    const d = new Date(f.fecha + 'T12:00:00');
    return EMISOR.lugar + ', ' + d.getDate() + ' de ' +
      ['Enero','Febrero','Marzo','Abril','Mayo','Junio','Julio','Agosto','Septiembre','Octubre','Noviembre','Diciembre'][d.getMonth()] +
      ' de ' + d.getFullYear();
  })();

  doc.setFont('helvetica', 'bold');
  doc.setFontSize(9);
  doc.setTextColor(...DARK);
  doc.text('FECHA:', MARGIN, 60);
  doc.setFont('helvetica', 'normal');
  doc.text(fechaStr, MARGIN + 15, 60);

  doc.setFont('helvetica', 'bold');
  doc.text('Nº FACTURA:', W - MARGIN - 50, 60);
  doc.setFont('helvetica', 'normal');
  doc.text(f.numero_factura || '—', W - MARGIN - 20, 60);

  // ── Sello PAGADO si procede ────────────────────────
  if (f.estado === 'Pagada') {
    doc.setFont('helvetica', 'bold');
    doc.setFontSize(18);
    doc.setTextColor(...[0, 150, 80]);
    doc.text('PAGADO', MARGIN, 72);
    doc.setLineWidth(1);
    doc.setDrawColor(...[0, 150, 80]);
    doc.line(MARGIN, 73.5, MARGIN + 28, 73.5);
  }

  // ── Bloque CLIENTE ─────────────────────────────────
  const clienteY = 78;
  doc.setFillColor(...ORANGE);
  doc.setDrawColor(...ORANGE);
  doc.roundedRect(MARGIN, clienteY, W - MARGIN * 2, 7, 1, 1, 'F');
  doc.setFont('helvetica', 'bold');
  doc.setFontSize(9);
  doc.setTextColor(255, 255, 255);
  doc.text('CLIENTE', MARGIN + 4, clienteY + 4.8);

  doc.setDrawColor(...BORDER);
  doc.setLineWidth(0.3);
  doc.rect(MARGIN, clienteY + 7, W - MARGIN * 2, 26);
  doc.setFont('helvetica', 'bold');
  doc.setFontSize(9);
  doc.setTextColor(...DARK);
  doc.text((f.receptor_nombre || '').toUpperCase(), MARGIN + 4, clienteY + 13);
  doc.setFont('helvetica', 'normal');
  doc.setFontSize(8.5);
  [
    f.receptor_dni ? 'D.N.I.: ' + f.receptor_dni : '',
    f.receptor_direccion || '',
    (f.receptor_cp ? 'C.P./' + f.receptor_cp + '  ' : '') + (f.receptor_municipio || ''),
  ].filter(Boolean).forEach((line, i) => doc.text(line, MARGIN + 4, clienteY + 18 + i * 4.5));

  // ── Tabla CONCEPTO ─────────────────────────────────
  let tableY = clienteY + 38;

  // Cabecera tabla
  doc.setFillColor(...ORANGE);
  doc.roundedRect(MARGIN, tableY, W - MARGIN * 2, 7, 1, 1, 'F');
  doc.setFont('helvetica', 'bold');
  doc.setFontSize(8.5);
  doc.setTextColor(255, 255, 255);
  const cols = [MARGIN + 4, 128, 150, 176];
  doc.text('CONCEPTO', cols[0], tableY + 4.8);
  doc.text('Nº HRS', cols[1], tableY + 4.8, { align: 'center' });
  doc.text('PRECIO / H', cols[2], tableY + 4.8, { align: 'center' });
  doc.text('TOTAL', W - MARGIN - 4, tableY + 4.8, { align: 'right' });

  tableY += 7;
  doc.setFont('helvetica', 'normal');
  doc.setFontSize(8.5);
  doc.setTextColor(...DARK);

  const lineas = Array.isArray(f.lineas) ? f.lineas : [];
  lineas.forEach((l, i) => {
    const rowH = 14;
    const bg = i % 2 === 0 ? [255,255,255] : CREAM;
    doc.setFillColor(...bg);
    doc.rect(MARGIN, tableY, W - MARGIN * 2, rowH, 'F');
    doc.setDrawColor(...BORDER);
    doc.line(MARGIN, tableY + rowH, W - MARGIN, tableY + rowH);

    // Concepto (con wrap)
    const lines = doc.splitTextToSize(l.concepto || '', 100);
    doc.setFont('helvetica', 'bold');
    doc.text(lines[0], cols[0], tableY + 5);
    if (lines[1]) {
      doc.setFont('helvetica', 'normal');
      doc.setFontSize(7.5);
      doc.text(lines[1], cols[0], tableY + 9.5);
      doc.setFontSize(8.5);
    }
    doc.setFont('helvetica', 'normal');
    doc.text(String(l.horas || 0) + ' h', cols[1], tableY + 5, { align: 'center' });
    doc.text(fmt2plain(l.precio_hora) + ' €/h', cols[2], tableY + 5, { align: 'center' });
    doc.setFont('helvetica', 'bold');
    doc.text(fmt2plain(l.importe), W - MARGIN - 4, tableY + 5, { align: 'right' });
    doc.text('€', W - MARGIN - 1, tableY + 5, { align: 'right' });
    doc.setFont('helvetica', 'normal');
    tableY += rowH;
  });

  // Fila TOTAL concepto
  doc.setFillColor(...CREAM);
  doc.rect(MARGIN, tableY, W - MARGIN * 2, 8, 'F');
  doc.setFont('helvetica', 'bold');
  doc.setFontSize(9);
  doc.setTextColor(...DARK);
  doc.text('TOTAL', W - MARGIN - 36, tableY + 5.5, { align: 'right' });
  doc.text(fmt2plain(f.base_imponible) + ' €', W - MARGIN - 4, tableY + 5.5, { align: 'right' });
  tableY += 14;

  // ── Bloque IMPORTE ────────────────────────────────
  doc.setFillColor(...ORANGE);
  doc.roundedRect(MARGIN, tableY, W - MARGIN * 2, 7, 1, 1, 'F');
  doc.setFont('helvetica', 'bold');
  doc.setFontSize(9);
  doc.setTextColor(255, 255, 255);
  doc.text('IMPORTE', MARGIN + 4, tableY + 4.8);
  tableY += 7;

  const importeRows = [
    ['Base imponible', fmt2plain(f.base_imponible) + ' €'],
    ['I.V.A.',         fmt2plain(f.iva_importe)    + ' €'],
    ['Retención Aplicable I.R.P.F', fmt2plain(f.irpf_importe) + ' €'],
  ];
  importeRows.forEach(([label, val], i) => {
    const rowH = 7;
    doc.setFillColor(i % 2 === 0 ? 255 : ...CREAM);
    doc.rect(MARGIN, tableY, W - MARGIN * 2, rowH, 'F');
    doc.setDrawColor(...BORDER);
    doc.line(MARGIN, tableY + rowH, W - MARGIN, tableY + rowH);
    doc.setFont('helvetica', i === 1 || i === 2 ? 'bolditalic' : 'normal');
    doc.setFontSize(8.5);
    doc.setTextColor(...DARK);
    doc.text(label, MARGIN + 4, tableY + 5);
    doc.text(val, W - MARGIN - 4, tableY + 5, { align: 'right' });
    tableY += rowH;
  });

  // Fila TOTAL FACTURA
  doc.setFillColor(240, 240, 250);
  doc.rect(MARGIN, tableY, W - MARGIN * 2, 9, 'F');
  doc.setFont('helvetica', 'bold');
  doc.setFontSize(10);
  doc.setTextColor(...DARK);
  doc.text('TOTAL FACTURA', MARGIN + 4, tableY + 6.2);
  doc.setTextColor(...BLUE);
  doc.text(fmt2plain(f.total) + ' €', W - MARGIN - 4, tableY + 6.2, { align: 'right' });
  tableY += 14;

  // ── IBAN + nota legal ──────────────────────────────
  doc.setFont('helvetica', 'normal');
  doc.setFontSize(8);
  doc.setTextColor(...DARK);
  doc.text('Nº Cuenta: ' + EMISOR.iban, W / 2, tableY, { align: 'center' });
  tableY += 5;
  if (f.notas) {
    doc.setTextColor(...MUTED);
    doc.setFontSize(7.5);
    const notaLines = doc.splitTextToSize('* ' + f.notas, W - MARGIN * 2);
    doc.text(notaLines, W / 2, tableY, { align: 'center' });
    tableY += notaLines.length * 4;
  }

  // ── Línea azul pie ────────────────────────────────
  doc.setFillColor(...BLUE);
  doc.rect(0, 295, W, 2, 'F');

  // ── Exportar ──────────────────────────────────────
  const filename = 'IPSE_Factura_' + (f.numero_factura || f.id_factura || 'nueva').replace(/\//g, '-') + '.pdf';
  if (descargar) {
    doc.save(filename);
    toast('PDF descargado ✓', 'ok');
  } else {
    const blob = doc.output('blob');
    window.open(URL.createObjectURL(blob), '_blank');
  }
}

// ── Export XLSX ────────────────────────────────────────
function exportarFacturasXLSX() {
  const data = [['Nº Factura','Fecha','Paciente','Receptor','Base imp.','IVA %','IVA €','IRPF %','IRPF €','Total','Estado']];
  (G.facturas || []).forEach(f => data.push([
    f.numero_factura, f.fecha, nomPacFull(f.id_paciente),
    f.receptor_nombre, f.base_imponible, f.iva_pct, f.iva_importe,
    f.irpf_pct, f.irpf_importe, f.total, f.estado
  ]));
  descargarXLSX(data, 'facturas_ipse.xlsx');
}
